**THE SCRIPT:**

#!/usr/bin/env bash

pwd_list="dictionary.txt"

while IFS= read -r pwd; do
  echo "testing password: $pwd"

  output=$(python3 decode.py <<< "$pwd")

  if [[ "$output" == *"That password is incorrect"* ]]; then
    echo "Password '$pwd' is incorrect"
  else
    echo "Correct pwd found: '$pwd'"
    echo "Program output: $output"
    break
  fi  

done < "$pwd_list"
